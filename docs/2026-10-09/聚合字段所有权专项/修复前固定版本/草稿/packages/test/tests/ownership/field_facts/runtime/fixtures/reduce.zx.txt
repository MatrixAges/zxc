export type Input = { values: u64[], count: u64, choice: bool }

export type Output = { fresh: u64[], view: u64[] }

export default function (in: Input): Output {
    const fresh: u64[] = [71]

    const initial = { fresh: fresh, view: in.values }

    const result = in.values.reduce((state, item) => { fresh: state.fresh.push(item)[0], view: state.view }, initial)

    return { fresh: result.fresh, view: result.view }
}
