export type Input = { values: u64[], count: u64, choice: bool }

export type Output = { fresh: u64[], view: u64[] }

export default function (in: Input): Output {
    const fresh: u64[] = [71]

    const untouched: u64[] = [83]

    const result = loop({ v0: fresh, v1: fresh, v2: fresh, stable: untouched, source: in.values, index: 0, limit: in.count }, {
        while: state => state.index < state.limit,
        next: state => {
            state = {
                v0: state.v1,
                v1: state.v2,
                v2: state.source,
                stable: state.stable,
                source: state.source,
                index: state.index + 1,
                limit: state.limit
            }
        }
    })

    return { fresh: result.stable, view: result.v0 }
}
