const std = @import("std");
const program = @import("program");
const host = @import("host");
const Visit = struct { previous: i64, current: i64, index: u64, source_value: i64, source_length: u64 };

fn check(args: struct {
    input: struct { items: []const i64, seed: i64 },
    failure: host.Failure,
    fail_at: usize,
    expected: struct {
        calls: usize,
        visits: []const Visit,
        events: []const u8,
        source_calls: usize,
        seed_calls: usize,
        result: union(enum) { value: i64, failure: anyerror },
    },
}) !void {
    for (0..2) |_| {
        var storage: [8194]i64 = @splat(-987654321);

        @memcpy(storage[1..][0..args.input.items.len], args.input.items);

        const original = storage;
        const input = @typeInfo(program.Input).pointer.child{ .items = storage[1..][0..args.input.items.len], .seed = args.input.seed };

        host.reset(args.failure, args.fail_at);

        var arena = std.heap.ArenaAllocator.init(std.testing.allocator);

        defer arena.deinit();

        const actual = program.execute(&arena, &input);

        try std.testing.expectEqualSlices(i64, &original, &storage);
        try std.testing.expectEqual(args.input.seed, input.seed);
        try std.testing.expectEqual(args.expected.source_calls, host.source_calls);
        try std.testing.expectEqual(args.expected.seed_calls, host.seed_calls);
        try std.testing.expectEqual(args.expected.calls, host.count);
        try std.testing.expectEqualSlices(u8, args.expected.events, host.events[0..host.event_count]);

        for (args.expected.visits, host.visits[0..host.count]) |expected, visit| {
            try std.testing.expectEqual(expected.previous, visit.previous);
            try std.testing.expectEqual(expected.current, visit.current);
            try std.testing.expectEqual(expected.index, visit.index);
            try std.testing.expectEqual(expected.source_length, visit.source.len);
            try std.testing.expect(visit.source.ptr == input.items.ptr);
            try std.testing.expectEqualSlices(i64, input.items, visit.source);
            try std.testing.expect(visit.index < visit.source.len);
            try std.testing.expectEqual(expected.source_value, visit.source[@intCast(visit.index)]);
        }

        switch (args.expected.result) {
            .value => |value| try std.testing.expectEqual(value, try actual),
            .failure => |failure| try std.testing.expectError(failure, actual),
        }
    }
}
