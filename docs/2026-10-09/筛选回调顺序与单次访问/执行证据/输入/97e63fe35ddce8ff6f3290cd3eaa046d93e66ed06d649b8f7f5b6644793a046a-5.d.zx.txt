export type Entry = { key: string
 value: string }
export type ParseOptions = { query: string
 separator: string
 assignment: string
 max_keys: u32 }
export type StringifyOptions = { entries: Entry[]
 separator: string
 assignment: string }

export declare function escape(allocator, in: string): string throws { InvalidUtf8, OutOfMemory } concurrent

export declare function unescape(allocator, in: string): string throws { OutOfMemory } concurrent

export declare function parse(allocator, in: string): Entry[] throws { OutOfMemory } concurrent

export declare function parseWith(allocator, in: ParseOptions): Entry[] throws { OutOfMemory } concurrent

export declare function stringify(allocator, in: Entry[]): string throws { InvalidUtf8, OutOfMemory } concurrent

export declare function stringifyWith(allocator, in: StringifyOptions): string throws { InvalidUtf8, OutOfMemory } concurrent
