import type { Json } from './shared/json.ts'
import { resolve } from 'node:path'
import { package_dir, writeCatalog, writeOutput } from './shared/catalog.ts'
import { readRows } from './shared/json.ts'

type Method = 'every' | 'some' | 'map'
type Probe = {
    rule: 'above' | 'equal' | 'always' | 'before' | 'after'
    threshold: number
    value: boolean
    failure: number
}
type Sample = {
    name: string
    method: Method
    path: string
    sha256: string
    input: Array<number>
    probe: Probe
    reason: string
    existing: boolean
}
type Expected = {
    value?: boolean | Array<boolean>
    error?: string
    calls: number
    visited: Array<number>
    input: Array<number>
}
type Row = { id: string; input: Array<number>; probe: Probe; expected: Expected }

const samples = readRows<Sample>(resolve(package_dir, 'src/data/predicate_reviews.jsonl'))
const base = 'built_ins/list/predicates/observations'
const inputs = [
    { name: 'empty', input: [] },
    { name: 'below', input: [9] },
    { name: 'above', input: [11] },
    { name: 'threshold', input: [10] },
    { name: 'false_true', input: [9, 11] },
    { name: 'true_false', input: [11, 9] },
    { name: 'all_true', input: [11, 12, 13] },
    { name: 'all_false', input: [0, -1, 9] },
    { name: 'ordinal_control', input: [99, -8, 44, 0, 12, 12, -5, 7, 1, 900] },
    { name: 'repeated', input: [11, 11, 11] },
    { name: 'growth_257', input: Array.from({ length: 257 }, (_, index) => index % 19) }
]
const probes: Array<{ name: string; probe: Probe }> = [
    { name: 'above_10', probe: { rule: 'above', threshold: 10, value: false, failure: 0 } },
    { name: 'equal_11', probe: { rule: 'equal', threshold: 11, value: false, failure: 0 } },
    { name: 'constant_false', probe: { rule: 'always', threshold: 0, value: false, failure: 0 } },
    { name: 'constant_true', probe: { rule: 'always', threshold: 0, value: true, failure: 0 } },
    { name: 'before_5', probe: { rule: 'before', threshold: 5, value: false, failure: 0 } },
    { name: 'after_5', probe: { rule: 'after', threshold: 5, value: false, failure: 0 } }
]

function expectation(args: { method: Method; input: Array<number>; probe: Probe }): Expected {
    const { method, input, probe } = args
    const visited: Array<number> = []

    try {
        const value = input[method]((item, index) => {
            visited.push(item)

            if (probe.failure !== 0 && visited.length === probe.failure) throw new Error('NativeFailure')
            if (probe.rule === 'above') return item > probe.threshold
            if (probe.rule === 'equal') return item === probe.threshold
            if (probe.rule === 'before') return index <= probe.threshold
            if (probe.rule === 'after') return index > probe.threshold

            return probe.value
        })

        return { value, calls: visited.length, visited, input }
    } catch (error) {
        if (!(error instanceof Error) || error.message !== 'NativeFailure') throw error

        return { error: 'NativeFailure', calls: visited.length, visited, input }
    }
}

const originals = new Map<string, Row>()

for (const method of ['every', 'some', 'map'] as const) {
    const rows: Array<Row> = samples
        .filter(sample => sample.method === method)
        .map(sample => {
            const row = {
                id: `${base}/${method}/${sample.name}`,
                input: sample.input,
                probe: sample.probe,
                expected: expectation(sample)
            }

            originals.set(sample.path, row)

            return row
        })

    for (const probe of probes) {
        if (method === 'map' && ['before', 'after'].includes(probe.probe.rule)) continue

        for (const input of inputs) {
            rows.push({
                id: `${base}/${method}/${probe.name}/${input.name}`,
                input: input.input,
                probe: probe.probe,
                expected: expectation({ method, input: input.input, probe: probe.probe })
            })
        }
    }

    for (const failure of [1, 2, 3, 6, 7, 10, 11]) {
        for (const value of [false, true]) {
            const input = [9, 8, 7, 6, 5, 4, 3, 2, 1, 0]
            const probe: Probe = { rule: 'always', threshold: 0, value, failure }

            rows.push({
                id: `${base}/${method}/failure_${failure}_${value}`,
                input,
                probe,
                expected: expectation({ method, input, probe })
            })
        }
    }

    const output_type = method === 'map' ? 'bool[]' : 'bool'

    writeOutput(
        `tests/${base}/${method}.zx`,
        `import host from "zig:host"\n\nexport type Input = i64[]\n\nexport type Output = ${output_type}\n\nexport default function (in: Input): Output {\n    return in.${method}(item => host.predicate(item))\n}\n`
    )
    writeCatalog(`tests/${base}/${method}.jsonl`, rows)
}

writeCatalog(
    'upstream/reviews/built_ins/array/predicate_native.jsonl',
    samples
        .filter(sample => !sample.existing)
        .map(sample => {
            const row = originals.get(sample.path)!

            return {
                path: sample.path,
                sha256: sample.sha256,
                status: 'adapted',
                reason: sample.reason,
                contract: 'packages/core/IR契约.md#所有权与集合回调',
                cases: [row.id],
                assertions: Object.entries<Json>(row.expected).map(([field, expected]) => ({
                    case: row.id,
                    field,
                    expected
                }))
            }
        })
)
