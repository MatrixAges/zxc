const std = @import("std");
const program = @import("program");

fn observe(count: u64, forbid_resize: bool) !void {
    const values = try std.testing.allocator.alloc(u64, 65537);

    defer std.testing.allocator.free(values);

    for (values, 0..) |*value, index| value.* = @intCast(index + 1);

    var tracked = std.testing.FailingAllocator.init(std.testing.allocator, .{ .resize_fail_index = if (forbid_resize) 0 else std.math.maxInt(usize) });

    {
        var arena = std.heap.ArenaAllocator.init(tracked.allocator());

        defer arena.deinit();

        const output = try program.execute(&arena, &.{ .values = values, .count = count });

        try std.testing.expectEqual(@as(u64, 2) + count, output.values[0]);
        try std.testing.expectEqual(@as(u64, 1), values[0]);

        std.debug.print("rounds={d} forbid_resize={} bytes={d} allocations={d} resizes={d} capacity={d}\n", .{ count, forbid_resize, tracked.allocated_bytes, tracked.allocations, tracked.resize_index, arena.queryCapacity() });
    }

    try std.testing.expectEqual(tracked.allocated_bytes, tracked.freed_bytes);
}

test "observe identical filter invocations before attributing costs to loop counts" {
    for ([_]bool{ false, true }) |forbid_resize| {
        for ([_]u64{ 1, 1, 64, 64, 1, 64 }) |count| try observe(count, forbid_resize);
    }
}
