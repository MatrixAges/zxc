export type Input = { values: u64[], count: u64 }

export type Output = { original: u64[], values: u64[] }

export default function (in: Input): Output {
    const fresh = in.values.map(item => item + 1)
    const initial = { values: in.values, round: 0 }

    const state = loop(initial, {
        while: current => current.round < in.count,
        next: current => {
            const inner = loop({ values: fresh, round: 0 }, {
                while: step => step.round < 3,
                next: step => {
                    step.values[0] += 1
                    step.round += 1
                }
            })

            current.values = inner.values
            current.round += 1
        }
    })

    return { original: in.values, values: state.values }
}
