export type Input = { values: u64[], count: u64 }

export type Output = { original: u64[], values: u64[] }

export default function (in: Input): Output {
    const initial = { pair: [in.values.map(item => item + 1), 0] }

    const state = loop(initial, {
        while: current => current.pair[1] < in.count,
        next: current => {
            current.pair[0][0] += 1
            current.pair[1] += 1
        }
    })

    return { original: in.values, values: state.pair[0] }
}
