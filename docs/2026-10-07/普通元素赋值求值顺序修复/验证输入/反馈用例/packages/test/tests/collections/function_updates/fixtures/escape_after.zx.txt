import type { PatchInput, PatchOutput } from "./types"

export type Input = PatchInput

export type Output = PatchOutput

export default function (in: Input): Output {
    const seen: i64 = 0

    const initial = { ...in, round: 0, boxes: [{ values: in.values }], seen }

    const result = loop(initial, {
        while: state => state.round < state.count,
        next: state => {
            state.values[state.selected] += state.delta

            state.boxes[0] = { values: state.values }

            state.seen += state.boxes[0].values[state.selected]
            state.round += 1
        }
    })

    return { values: result.values, mirror: result.boxes[0].values, seen: result.seen }
}
