export type Input = { values: i64[], mirror: i64[], count: u64, selected: u64, delta: i64 }

export type Output = { values: i64[], mirror: i64[], seen: i64 }

export default function (in: Input): Output {
    const initial = { ...in, round: 0 }

    const result = loop(initial, {
        while: state => state.round < state.count,
        next: state => {
            state.values[state.selected] += state.delta
            state.mirror[state.selected] += state.delta * 2

            state.round += 1
        }
    })

    return { values: result.values, mirror: result.mirror, seen: 0 }
}
