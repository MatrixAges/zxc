const std = @import("std");
const reference = @import("../references/ir/fixture.zig");
pub const compiler = @import("compiler");
pub const checks = @import("checks");
pub const ir = compiler.ir;

comptime {
    if (!checks.generated) @compileError("native module checks require the generated parser path");
}

pub const Case = struct {
    analysis: compiler.AnalysisResult,
    arena: std.heap.ArenaAllocator,
    program: ir.Program,
    pub fn init() !Case {
        const analysis = try reference.analyze(std.testing.allocator);

        return .{ .analysis = analysis, .arena = std.heap.ArenaAllocator.init(std.testing.allocator), .program = analysis.value.ir };
    }
    pub fn deinit(self: *Case) void {
        self.arena.deinit();
        self.analysis.deinit();
    }

    pub fn modules(self: *Case, rows: []const ir.NativeModule) !void {
        self.program.native_modules = try ir.NativeModuleTable.fromValues(self.arena.allocator(), rows);

        try std.testing.expect(self.program.native_modules.validStructure());
    }

    pub fn functions(self: *Case, rows: []const ir.Function) !void {
        self.program.functions = try ir.FunctionTable.fromValues(self.arena.allocator(), rows);

        try std.testing.expect(self.program.functions.validStructure());
    }

    pub fn nativeFunction(self: *const Case) ir.Function {
        for (0..self.program.functions.count()) |index| {
            if (self.program.functions.native_modules[index] != null) return self.program.functions.at(index);
        }

        unreachable;
    }

    pub fn expectModules(self: *const Case, expected: bool) !void {
        try std.testing.expectEqual(expected, checks.native.validate(self.program));
    }
    pub fn expectExport(self: *const Case, index: usize, expected: bool) !void {
        try std.testing.expect(self.program.functions.validStructure());
        try std.testing.expect(index < self.program.functions.count());
        try std.testing.expect(self.program.native_modules.validStructure());

        const item = self.program.functions.at(index);

        if (item.external) |external| {
            try std.testing.expect(@backingInt(external.module) < self.program.native_modules.count());
            try std.testing.expect(external.member.len != 0);
        }

        try std.testing.expectEqual(expected, checks.native.validateExport(self.program, index));
    }
};
