const std = @import("std");
const compiler = @import("compiler");
const importer = @import("../parallel/library/import.zig");
const Output = @import("library_output").Output;

fn restore(init: std.process.Init, bytes: []u8, route: []const u8, directory: []const u8) !compiler.library.Result {
    const allocator = init.arena.allocator();
    const temporary = std.heap.page_allocator;
    var output = Output{ .io = init.io, .allocator = allocator, .directory = directory };

    try output.file("workflow.zxlib", bytes);

    var library = try compiler.library.codec.decode(temporary, bytes);

    defer library.deinit();

    const encoded = try compiler.library.codec.encode(allocator, &library);

    defer allocator.free(encoded);

    try std.testing.expectEqualSlices(u8, bytes, encoded);

    @memset(bytes, 0xdd);

    const name = if (std.mem.endsWith(u8, route, "reverse")) "alias" else "workflow";

    const index = for (library.exports, 0..) |exported, index| {
        if (std.mem.eql(u8, exported.name, name)) break index;
    } else return error.MissingExport;

    if (std.mem.eql(u8, route, "forward") or std.mem.eql(u8, route, "reverse")) {
        _ = try library.module(index);

        return compiler.library.codec.decode(allocator, encoded);
    }

    var consumer = block: {
        var imported = try importer.link(temporary, &library, name, if (std.mem.startsWith(u8, route, "zx_")) .zx else .rx);

        defer imported.deinit();

        const serialized = try compiler.library.codec.encode(allocator, &imported);

        defer allocator.free(serialized);

        try output.file("consumer.zxlib", serialized);

        const decoded = try compiler.library.codec.decode(allocator, serialized);

        @memset(serialized, 0xdd);

        break :block decoded;
    };

    defer consumer.deinit();

    if (std.mem.startsWith(u8, route, "republish_")) {
        var republished = try importer.link(allocator, &consumer, "consumer", .zx);

        defer republished.deinit();

        const serialized = try compiler.library.codec.encode(allocator, &republished);

        defer allocator.free(serialized);

        try output.file("republished.zxlib", serialized);

        const decoded = try compiler.library.codec.decode(allocator, serialized);

        @memset(serialized, 0xdd);

        return decoded;
    }

    const serialized = try compiler.library.codec.encode(allocator, &consumer);

    defer allocator.free(serialized);

    const decoded = try compiler.library.codec.decode(allocator, serialized);

    @memset(serialized, 0xdd);

    return decoded;
}

pub fn emit(init: std.process.Init, bytes: []u8, route: []const u8, directory: []const u8) !void {
    var restored = try restore(init, bytes, route, directory);

    defer restored.deinit();

    const name = if (std.mem.eql(u8, route, "forward")) "workflow" else if (std.mem.eql(u8, route, "reverse")) "alias" else "consumer";

    const index = for (restored.exports, 0..) |exported, index| {
        if (std.mem.eql(u8, exported.name, name)) break index;
    } else return error.MissingExport;

    try @import("output.zig").emit(init, try restored.module(index), directory);
}
