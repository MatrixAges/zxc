const std = @import("std");
const Input = @import("program").Input;

pub const Storage = struct {
    items: [1026]i64,
    replacement: [5]i64,
    data: Input,
    pub fn init(data: Input) Storage {
        var result = Storage{ .items = @splat(-9191), .replacement = @splat(7373), .data = data };

        @memcpy(result.items[1..][0..data.items.len], data.items);
        @memcpy(result.replacement[1..][0..data.replacement.len], data.replacement);

        return result;
    }
    pub fn input(self: *Storage) Input {
        var result = self.data;

        result.items = self.items[1..][0..self.data.items.len];
        result.replacement = self.replacement[1..][0..self.data.replacement.len];

        return result;
    }
    pub fn check(self: *const Storage, before: *const Storage, actual: Input, original: Input) !void {
        try std.testing.expectEqualSlices(i64, &before.items, &self.items);
        try std.testing.expectEqualSlices(i64, &before.replacement, &self.replacement);
        try std.testing.expectEqual(original.items.ptr, actual.items.ptr);
        try std.testing.expectEqual(original.replacement.ptr, actual.replacement.ptr);
        try std.testing.expectEqual(original.items.len, actual.items.len);
        try std.testing.expectEqual(original.replacement.len, actual.replacement.len);
        try std.testing.expectEqual(original.index, actual.index);
        try std.testing.expectEqual(original.count, actual.count);
        try std.testing.expectEqual(original.value, actual.value);
    }
};

pub fn independent(left: []const i64, right: []const i64) !void {
    if (left.len == 0 or right.len == 0) return;

    const left_start = @intFromPtr(left.ptr);
    const right_start = @intFromPtr(right.ptr);

    try std.testing.expect(left_start + left.len * @sizeOf(i64) <= right_start or right_start + right.len * @sizeOf(i64) <= left_start);
}
