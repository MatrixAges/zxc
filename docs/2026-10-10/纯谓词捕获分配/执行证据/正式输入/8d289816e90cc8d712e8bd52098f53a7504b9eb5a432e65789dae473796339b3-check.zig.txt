const std = @import("std");
const program = @import("program");
const options = @import("options");

pub const Interface = enum { pointer, value };

pub fn expected(values: []const i64, threshold: i64) bool {
    for (values) |item| {
        const passed = item > threshold;

        if (options.universal and !passed) return false;
        if (!options.universal and passed) return true;
    }

    return options.universal;
}

pub fn observe(comptime interface: Interface, count: usize, pattern: usize, deny: bool) !bool {
    var storage: [4098]i64 = @splat(-65535);

    for (storage[1..][0..count], 0..) |*item, index| {
        item.* = switch (pattern) {
            0 => 11,
            1 => -3,
            else => @as(i64, @intCast(index % 17)) - 4,
        };
    }

    const snapshot = storage;
    const values = storage[1..][0..count];
    const threshold: i64 = if (options.captured_threshold) 7 else 0;
    var meter = std.testing.FailingAllocator.init(std.testing.allocator, .{ .fail_index = if (deny) 0 else std.math.maxInt(usize) });
    var arena = std.heap.ArenaAllocator.init(meter.allocator());

    defer arena.deinit();

    const Parameter = if (interface == .pointer) program.Input else @typeInfo(@TypeOf(program.executeValue)).@"fn".param_types[1].?;
    const indirect = @typeInfo(Parameter) == .pointer;
    const Input = if (indirect) std.meta.Child(Parameter) else Parameter;
    const input: Input = .{ .items = values, .threshold = 7 };
    const actual = if (interface == .pointer) program.execute(&arena, &input) else program.executeValue(meter.allocator(), if (indirect) &input else input);
    const succeeded = if (actual) |value| block: {
        try std.testing.expectEqual(expected(values, threshold), value);

        break :block true;
    } else |failure| block: {
        try std.testing.expect(deny);
        try std.testing.expectEqual(error.OutOfMemory, failure);

        break :block false;
    };

    try std.testing.expectEqualSlices(i64, &snapshot, &storage);
    try std.testing.expectEqual(@as(i64, 7), input.threshold);
    try std.testing.expectEqual(values.ptr, input.items.ptr);
    try std.testing.expectEqual(values.len, input.items.len);

    std.debug.print("ALLOCATION interface={t} count={d} pattern={d} deny={} succeeded={} allocations={d} allocated_bytes={d} induced={}\n", .{ interface, count, pattern, deny, succeeded, meter.allocations, meter.allocated_bytes, meter.has_induced_failure });

    return succeeded and meter.allocations == 0 and !meter.has_induced_failure;
}
