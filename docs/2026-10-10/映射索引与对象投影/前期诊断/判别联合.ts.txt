type Items = { items: Array<number> }
export type Spec =
    | { operation: 'source_value'; input: Items & { threshold: number } }
    | { operation: 'indices' | 'length' | 'rotate'; input: Items }
    | { operation: 'records'; input: Items & { first: number } }

export default function expectation(spec: Spec) {
    const mapped = spec.input.items.map((item, index, source) => {
        if (spec.operation === 'source_value') return item > spec.input.threshold && source[index] === item
        if (spec.operation === 'indices') return index
        if (spec.operation === 'length') return source.length
        if (spec.operation === 'rotate') return source[(index + 1) % source.length]

        return { item, index, length: source.length, previous: index === 0 ? spec.input.first : source[index - 1] }
    })

    return { value: { mapped, items: spec.input.items } }
}
