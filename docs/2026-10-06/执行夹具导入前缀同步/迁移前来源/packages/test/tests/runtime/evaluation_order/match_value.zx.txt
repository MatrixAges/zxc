import readLeft from "lib:probe-left" 
import readRight from "lib:probe-right" 

export type Input = { scale: f64
 fail_left: bool
 fail_right: bool
 items: f64[]  } 

export type Output = f64 

export default function (in: Input): Output {
  return match (readLeft(in.fail_left) * in.scale) {
    readRight(in.fail_right) => in.items[0],
    readLeft(false) => in.items[1],
    _ => in.items[2],
  } 
}
