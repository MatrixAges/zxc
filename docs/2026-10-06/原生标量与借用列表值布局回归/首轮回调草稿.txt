import advance from "./borrowed_objects/loop"

export type Cell = { value: u64, label: string }

export type State = { index: u64, total: u64, values: Cell[], saved: Cell? }

export type Input = { values: Cell[], seed: u64 }

export type Output = { empty: State, results: { first: State, second: State }[] }

export default function (in: Input): Output {
  const empty = advance({ index: in.values.length, total: in.seed, values: in.values, saved: null })
  const results = in.values.map((cell, index) => {
    const first = advance({ index: 0, total: in.seed + index * 2, values: in.values, saved: cell })
    const second = advance({ index: 0, total: in.seed + index * 2 + 1, values: in.values, saved: cell })

    return { first: first, second: second }
  })

  return { empty: empty, results: results }
}
