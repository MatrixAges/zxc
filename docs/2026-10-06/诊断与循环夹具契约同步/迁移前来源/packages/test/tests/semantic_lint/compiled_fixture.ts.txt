import assert from 'node:assert/strict'
import { spawnSync } from 'node:child_process'
import { readFileSync } from 'node:fs'
import { join, resolve } from 'node:path'
import createFixture from './fixture.ts'

export const identity = `export type Input = u64

export type Output = u64

export default function (in: Input): Output {
  return in
}
`

export default function publishLibrary() {
	const fixture = createFixture({
		'pkg.yaml': `name: bundle
version: 1.0.0

exports:
    .: identity.zx
    ./flow: flow.rx
`,
		'identity.zx': identity,
		'flow.rx': `<Module>
  <Call fn="identity" in={$in} name="result" />

  <Return value={ctx.result} />
</Module>
`
	})

	try {
		const result = spawnSync(
			resolve(process.argv[2]),
			['build', 'pkg.yaml', '--mode', 'lib', '--out', 'published'],
			{
				cwd: fixture.root,
				encoding: 'utf8',
				timeout: 180_000,
				env: { ...process.env, ZXC_CACHE_DIR: join(fixture.root, 'cache') }
			}
		)
		assert.ifError(result.error)
		assert.equal(result.signal, null, result.stderr)
		assert.equal(result.status, 0, result.stderr)

		return {
			'pkgs/bundle/pkg.yaml': readFileSync(join(fixture.root, 'published/pkg.yaml'), 'utf8'),
			'pkgs/bundle/library.zxcir': readFileSync(join(fixture.root, 'published/library.zxcir'))
		}
	} finally {
		fixture.cleanup()
	}
}
