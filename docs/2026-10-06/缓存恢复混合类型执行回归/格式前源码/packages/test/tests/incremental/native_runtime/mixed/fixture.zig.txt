const std = @import("std");
const program = @import("program");
const host = @import("host");
const Input = std.meta.Child(program.Input);
const Record = std.meta.Child(@FieldType(Input, "record"));
const Pair = std.meta.Child(@FieldType(Input, "pair"));

pub const Case = struct {
    values: []const host.Mode,
    saved: ?u64,
    record_value: u64,
    record_enabled: bool,
    pair_value: u64,
    pair_enabled: bool,
    mode: host.Mode,
    index: u64,
    output_value: u64,
    picked: ?host.Mode,
    missing: bool,
};

pub const Fixture = struct {
    case: Case,
    values: []host.Mode,
    record: Record,
    pair: Pair,

    pub fn init(case: Case) !Fixture {
        return .{
            .case = case,
            .values = try std.testing.allocator.dupe(host.Mode, case.values),
            .record = .{ .value = case.record_value, .enabled = case.record_enabled },
            .pair = .{ case.pair_value, case.pair_enabled },
        };
    }

    pub fn deinit(self: Fixture) void {
        std.testing.allocator.free(self.values);
    }

    pub fn input(self: *const Fixture) Input {
        return .{ .values = self.values, .saved = self.case.saved, .record = &self.record, .pair = &self.pair, .mode = self.case.mode, .index = self.case.index };
    }

    pub fn checkInput(self: *const Fixture, value: Input) !void {
        try std.testing.expectEqualSlices(host.Mode, self.case.values, value.values);
        try std.testing.expectEqual(self.values.len, value.values.len);
        try std.testing.expect(value.values.ptr == self.values.ptr);
        try std.testing.expect(value.record == &self.record and value.pair == &self.pair);
        try std.testing.expectEqual(self.case.saved, value.saved);
        try std.testing.expectEqual(self.case.mode, value.mode);
        try std.testing.expectEqual(self.case.index, value.index);
        try std.testing.expectEqual(self.case.record_value, value.record.value);
        try std.testing.expectEqual(self.case.record_enabled, value.record.enabled);
        try std.testing.expectEqual(self.case.pair_value, value.pair.@"0");
        try std.testing.expectEqual(self.case.pair_enabled, value.pair.@"1");
    }

    pub fn checkOutput(self: *const Fixture, value: program.Output) !void {
        try std.testing.expectEqualSlices(host.Mode, self.case.values, value.values);
        if (value.values.len != 0) try std.testing.expect(value.values.ptr == self.values.ptr);
        try std.testing.expectEqual(self.case.saved, value.saved);
        try std.testing.expectEqual(self.case.mode, value.mode);
        try std.testing.expectEqual(self.case.picked, value.picked);
        try std.testing.expectEqual(self.case.missing, value.missing);
        try std.testing.expectEqual(self.case.output_value, value.record.value);
        try std.testing.expectEqual(self.case.pair_enabled, value.record.enabled);
        try std.testing.expectEqual(self.case.pair_value, value.pair.@"0");
        try std.testing.expectEqual(self.case.pair_enabled, value.pair.@"1");
    }
};

pub fn checkCalls(mode: host.Mode, count: usize) !void {
    const expected: [2]host.Mode = if (mode == .First) .{ .First, .Second } else .{ .Second, .First };

    try std.testing.expectEqual(count, host.calls);
    try std.testing.expectEqualSlices(host.Mode, expected[0..count], host.trace[0..count]);
}
