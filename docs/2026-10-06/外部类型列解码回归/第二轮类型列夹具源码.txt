const std = @import("std");

pub const source =
    \\export enum Phase { First, Second }
    \\export type Input = { index: u64, record: { value: u64 }, pair: [u64, bool], values: Phase[], saved: u64? }
    \\export type Output = Phase
    \\export default function (in: Input): Output { return in.values[in.index] }
;

pub const Mode = enum {
    first_length,
    second_length,
    labels_length,
    fields_length,
    unknown_kind,
    scalar_identity,
    scalar_second,
    illegal_label,
    optional_self,
    list_void,
    tuple_self,
    object_self,
    object_void,
    object_empty_name,
    object_unsorted,
    tuple_gap,
    tuple_overflow,
    object_gap,
    object_overflow,
    names_gap,
    names_overflow,
    enum_empty_label,
    enum_duplicate,
    error_empty_name,
};

fn column(table: *std.json.Value, name: []const u8) *std.json.Value {
    return table.object.getPtr(name).?;
}

fn row(table: *std.json.Value, kind: u8) !usize {
    const kinds = column(table, "kinds");
    const count = if (kinds.* == .string) kinds.string.len else kinds.array.items.len;

    for (0..count) |index| {
        const code = if (kinds.* == .string) kinds.string[index] else kinds.array.items[index].integer;

        if (code == kind and (kind != 1 or column(table, "second").array.items[index].integer >= 2)) return index;
    }

    return error.MissingFixtureKind;
}

fn value(table: *std.json.Value, name: []const u8, index: usize) *std.json.Value {
    return &column(table, name).array.items[index];
}

fn start(table: *std.json.Value, index: usize) usize {
    return @intCast(value(table, "first", index).integer);
}

pub fn corrupt(allocator: std.mem.Allocator, table: *std.json.Value, mode: Mode) !void {
    const object = try row(table, 1);
    const optional = try row(table, 2);
    const list = try row(table, 3);
    const tuple = try row(table, 4);
    const errors = try row(table, 5);
    const enumeration = try row(table, 7);

    try std.testing.expect(column(table, "children").array.items.len >= 2);
    try std.testing.expect(column(table, "field_names").array.items.len >= 2);
    try std.testing.expect(value(table, "second", enumeration).integer >= 2);
    try std.testing.expect(value(table, "second", errors).integer >= 1);

    switch (mode) {
        .first_length => column(table, "first").array.items.len -= 1,
        .second_length => column(table, "second").array.items.len -= 1,
        .labels_length => column(table, "labels").array.items.len -= 1,
        .fields_length => column(table, "field_names").array.items.len -= 1,
        .unknown_kind => {
            const kinds = column(table, "kinds");

            if (kinds.* == .string) {
                const codes = try allocator.dupe(u8, kinds.string);
                codes[0] = 9;
                kinds.* = .{ .string = codes };
            } else kinds.array.items[0] = .{ .integer = 9 };
        },
        .scalar_identity => value(table, "first", 0).* = .{ .integer = 1 },
        .scalar_second => value(table, "second", 0).* = .{ .integer = 1 },
        .illegal_label => value(table, "labels", tuple).* = .{ .string = "Unexpected" },
        .optional_self => value(table, "first", optional).* = .{ .integer = @intCast(optional) },
        .list_void => value(table, "first", list).* = .{ .integer = 0 },
        .tuple_self => value(table, "children", start(table, tuple)).* = .{ .integer = @intCast(tuple) },
        .object_self => value(table, "field_types", start(table, object)).* = .{ .integer = @intCast(object) },
        .object_void => value(table, "field_types", start(table, object)).* = .{ .integer = 0 },
        .object_empty_name => value(table, "field_names", start(table, object)).* = .{ .string = "" },
        .object_unsorted => std.mem.swap(std.json.Value, value(table, "field_names", start(table, object)), value(table, "field_names", start(table, object) + 1)),
        .tuple_gap => value(table, "first", tuple).integer += 1,
        .tuple_overflow => value(table, "second", tuple).* = .{ .integer = std.math.maxInt(u32) },
        .object_gap => value(table, "first", object).integer += 1,
        .object_overflow => value(table, "second", object).* = .{ .integer = std.math.maxInt(u32) },
        .names_gap => value(table, "first", enumeration).integer += 1,
        .names_overflow => value(table, "second", enumeration).* = .{ .integer = std.math.maxInt(u32) },
        .enum_empty_label => value(table, "labels", enumeration).* = .{ .string = "" },
        .enum_duplicate => value(table, "names", start(table, enumeration) + 1).* = value(table, "names", start(table, enumeration)).*,
        .error_empty_name => value(table, "names", start(table, errors)).* = .{ .string = "" },
    }
}
